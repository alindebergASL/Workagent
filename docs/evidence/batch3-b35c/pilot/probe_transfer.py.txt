def prefix_migrate(dsn):
 with Database(dsn).transaction() as c:
  c.execute('CREATE TABLE schema_migrations(name text PRIMARY KEY, checksum text NOT NULL, applied_at timestamptz NOT NULL DEFAULT now())')
  for entry in snapshot['migrations']:
   p=ROOT/'backend/migrations'/entry['name']
   assert hashlib.sha256(p.read_bytes()).hexdigest()==entry['checksum']
   c.execute(p.read_text())
   c.execute('INSERT INTO schema_migrations(name,checksum) VALUES (%s,%s)',(entry['name'],entry['checksum']))
